```javascript
async function randomScroll(session, options = {}) {
  const {
    minScrolls = 1,
    maxScrolls = 3,
    maxDistance = 500
  } = options;

  const scrollCount = Math.floor(random(minScrolls, maxScrolls + 1));
  
  for (let i = 0; i < scrollCount; i++) {
    await session.evaluate((distance) => {
      window.scrollBy({
        top: distance,
        behavior: 'smooth'
      });
    }, Math.floor(random(-maxDistance, maxDistance)));
    
    await randomDelay(500, 1500);
  }
}

// 添加鼠标随机移动
async function randomMouseMove(session, options = {}) {
  const { moves = 3 } = options;
  
  for (let i = 0; i < moves; i++) {
    await session.evaluate(() => {
      const x = Math.random() * window.innerWidth;
      const y = Math.random() * window.innerHeight;
      
      const event = new MouseEvent('mousemove', {
        view: window,
        bubbles: true,
        cancelable: true,
        clientX: x,
        clientY: y
      });
      
      document.dispatchEvent(event);
    });
    
    await randomDelay(100, 300);
  }
}
```

### 1. 智能等待机制优化

**当前问题：**
- 固定的 `sleep()` 延迟不够灵活
- 缺少元素状态的持续监控

**改进建议：**

```javascript
// 在 human-behavior.js 中添加智能等待函数
/**
 * 等待元素出现并可交互
 */
async function waitForElementReady(session, selector, options = {}) {
  const {
    timeout = 30000,
    checkInterval = 500,
    checkVisible = true,
    checkEnabled = true
  } = options;

  const deadline = Date.now() + timeout;
  
  while (Date.now() < deadline) {
    const result = await session.evaluate((sel, opts) => {
      const element = document.querySelector(sel);
      if (!element) return { ready: false, reason: '元素不存在' };
      
      const rect = element.getBoundingClientRect();
      const style = getComputedStyle(element);
      
      // 检查可见性
      if (opts.checkVisible) {
        if (rect.width === 0 || rect.height === 0) {
          return { ready: false, reason: '元素尺寸为0' };
        }
        if (style.display === 'none' || style.visibility === 'hidden' || style.opacity === '0') {
          return { ready: false, reason: '元素被隐藏' };
        }
      }
      
      // 检查是否启用
      if (opts.checkEnabled && element.disabled) {
        return { ready: false, reason: '元素被禁用' };
      }
      
      // 检查是否在视口内
      const inViewport = rect.top >= 0 && rect.top <= window.innerHeight;
      if (!inViewport) {
        return { ready: false, reason: '元素不在视口内' };
      }
      
      return { ready: true };
    }, selector, { checkVisible, checkEnabled });
    
    if (result.ready) return { ok: true };
    
    await sleep(checkInterval);
  }
  
  return { ok: false, reason: '等待元素就绪超时' };
}

/**
 * 等待页面加载完成
 */
async function waitForPageLoad(session, options = {}) {
  const { timeout = 30000 } = options;
  const deadline = Date.now() + timeout;
  
  while (Date.now() < deadline) {
    const state = await session.evaluate(() => {
      return {
        readyState: document.readyState,
        // 检查是否有加载中的资源
        hasLoadingResources: !!document.querySelector('[loading], .loading, .spinner'),
        // 检查 body 是否存在且有内容
        hasContent: document.body && document.body.children.length > 0
      };
    });
    
    if (state.readyState === 'complete' && !state.hasLoadingResources && state.hasContent) {
      return { ok: true };
    }
    
    await sleep(500);
  }
  
  return { ok: false, reason: '页面加载超时' };
}

/**
 * 等待网络空闲
 */
async function waitForNetworkIdle(session, options = {}) {
  const { timeout = 30000, idleTime = 500, maxInflight = 2 } = options;
  
  let inflightRequests = 0;
  let lastActivity = Date.now();
  const deadline = Date.now() + timeout;
  
  // 监听网络请求
  await session.command('Network.enable');
  
  const requestHandler = () => {
    inflightRequests++;
    lastActivity = Date.now();
  };
  
  const responseHandler = () => {
    inflightRequests = Math.max(0, inflightRequests - 1);
    lastActivity = Date.now();
  };
  
  // 注册监听器（需要在 NativeSession 中实现事件监听）
  // session.on('Network.requestWillBeSent', requestHandler);
  // session.on('Network.responseReceived', responseHandler);
  // session.on('Network.loadingFailed', responseHandler);
  // session.on('Network.loadingFinished', responseHandler);
  
  while (Date.now() < deadline) {
    const idle = inflightRequests <= maxInflight;
    const idleDuration = Date.now() - lastActivity;
    
    if (idle && idleDuration >= idleTime) {
      return { ok: true };
    }
    
    await sleep(100);
  }
  
  return { ok: false, reason: '等待网络空闲超时' };
}
```

### 2. 重试机制优化

**当前问题：**
- 重试策略相对简单
- 缺少失败原因分析

**改进建议：**

```javascript
/**
 * 智能重试包装器
 */
async function retryWithBackoff(fn, options = {}) {
  const {
    maxRetries = 3,
    baseDelay = 1000,
    maxDelay = 15000,
    backoffMultiplier = 2,
    shouldRetry = () => true,
    onRetry = () => {}
  } = options;
  
  let lastError;
  
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await fn(attempt);
    } catch (error) {
      lastError = error;
      
      // 检查是否应该重试
      if (attempt >= maxRetries || !shouldRetry(error, attempt)) {
        throw error;
      }
      
      // 计算延迟时间（指数退避）
      const delay = Math.min(
        baseDelay * Math.pow(backoffMultiplier, attempt - 1),
        maxDelay
      );
      
      // 添加随机抖动（±20%）
      const jitter = delay * 0.2 * (Math.random() * 2 - 1);
      const actualDelay = Math.floor(delay + jitter);
      
      onRetry(error, attempt, actualDelay);
      await sleep(actualDelay);
    }
  }
  
  throw lastError;
}

// 使用示例
async function robustQuery(session, company, query) {
  return retryWithBackoff(
    async (attempt) => {
      // 执行查询
      const result = await performQuery(session, company, query);
      return result;
    },
    {
      maxRetries: 3,
      shouldRetry: (error, attempt) => {
        // 根据错误类型决定是否重试
        const retryableErrors = [
          '页面加载超时',
          '元素不存在',
          '网络请求失败',
          '验证码识别失败'
        ];
        return retryableErrors.some(msg => error.message.includes(msg));
      },
      onRetry: (error, attempt, delay) => {
        console.log(`查询失败，${delay}ms 后进行第 ${attempt} 次重试：${error.message}`);
      }
    }
  );
}
```

```javascript
/**
 * 页面健康检查
 */
async function checkPageHealth(session) {
  return session.evaluate(() => {
    const checks = {
      documentReady: document.readyState === 'complete',
      hasBody: !!document.body,
      hasContent: document.body && document.body.children.length > 0,
      noErrors: !document.querySelector('.error, .error-page, #error'),
      notBlank: document.body && document.body.innerText.trim().length > 10
    };
    
    const healthy = Object.values(checks).every(v => v);
    
    return {
      healthy,
      checks,
      url: window.location.href,
      title: document.title
    };
  });
}

/**
 * 自动恢复机制
 */
async function autoRecover(session, url, maxAttempts = 3) {
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    const health = await checkPageHealth(session);
    
    if (health.healthy) {
      return { ok: true };
    }
    
    console.log(`页面状态异常（第 ${attempt} 次检查）:`, health.checks);
    
    if (attempt < maxAttempts) {
      // 尝试刷新页面
      await session.navigate(url);
      await waitForPageLoad(session);
      await randomDelay(2000, 4000);
    }
  }
  
  return { ok: false, reason: '页面恢复失败' };
}
```

```javascript
/**
 * 验证码智能处理
 */
async function smartCaptchaSolver(session, options = {}) {
  const {
    maxAttempts = 5,
    ocrRetries = 3,
    refreshOnFail = true
  } = options;
  
  let lastError;
  
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      // 等待验证码图片加载完成
      await waitForElementReady(session, captchaImageSelector, {
        timeout: 10000,
        checkVisible: true
      });
      
      // 多次尝试 OCR 识别
      let bestCode = null;
      let bestConfidence = 0;
      
      for (let ocrAttempt = 0; ocrAttempt < ocrRetries; ocrAttempt++) {
        const result = await recognizeCaptcha(session);
        
        if (result.confidence > bestConfidence) {
          bestCode = result.code;
          bestConfidence = result.confidence;
        }
        
        // 如果置信度足够高，直接使用
        if (result.confidence > 0.9) {
          break;
        }
        
        await randomDelay(500, 1000);
      }
      
      if (!bestCode) {
        throw new Error('验证码识别失败');
      }
      
      // 提交验证码
      await submitCaptchaWithDelay(session, bestCode);
      
      // 等待验证结果
      const outcome = await waitForCaptchaOutcome(session, { timeout: 5000 });
      
      if (outcome.passed) {
        return { ok: true };
      }
      
      lastError = '验证码错误';
      
      // 如果失败且允许刷新，刷新验证码
      if (refreshOnFail && attempt < maxAttempts) {
        await refreshCaptcha(session);
        await randomDelay(1000, 2000);
      }
      
    } catch (error) {
      lastError = error.message;
      console.log(`验证码处理失败（第 ${attempt} 次）：${error.message}`);
    }
  }
  
  return { ok: false, reason: lastError || '验证码处理失败' };
}
```

### 5. 异常情况检测和处理

**改进建议：**

```javascript
/**
 * 检测异常情况（反爬虫页面、错误页面等）
 */
async function detectAnomalies(session) {
  return session.evaluate(() => {
    const text = document.body.innerText.toLowerCase();
    const title = document.title.toLowerCase();
    
    const anomalies = {
      captchaChallenge: /验证码|captcha|robot|人机验证/.test(text),
      accessDenied: /access denied|拒绝访问|403|forbidden/.test(text),
      rateLimited: /too many requests|请求过于频繁|稍后再试/.test(text),
      serverError: /500|服务器错误|server error|内部错误/.test(text),
      networkError: /网络错误|network error|连接失败/.test(text),
      sessionExpired: /会话过期|session expired|重新登录/.test(text),
      maintenance: /维护中|maintenance|系统升级/.test(text)
    };
    
    const detected = Object.entries(anomalies)
      .filter(([_, value]) => value)
      .map(([key]) => key);
    
    return {
      hasAnomalies: detected.length > 0,
      detected,
      text: text.substring(0, 200)
    };
  });
}

/**
 * 异常处理策略
 */
async function handleAnomalies(session, anomalies, url) {
  if (!anomalies.hasAnomalies) {
    return { ok: true };
  }
  
  for (const anomaly of anomalies.detected) {
    switch (anomaly) {
      case 'rateLimited':
        // 速率限制：等待更长时间
        console.log('检测到访问频率限制，等待 30 秒...');
        await randomDelay(30000, 40000);
        await session.navigate(url);
        break;
        
      case 'sessionExpired':
        // 会话过期：清除 cookies 并重新开始
        console.log('会话过期，清除缓存并重试...');
        await session.command('Network.clearBrowserCookies');
        await session.command('Network.clearBrowserCache');
        await randomDelay(5000, 8000);
        await session.navigate(url);
        break;
        
      case 'accessDenied':
      case 'serverError':
        // 服务器错误：等待并重试
        console.log(`检测到 ${anomaly}，等待后重试...`);
        await randomDelay(10000, 15000);
        await session.navigate(url);
        break;
        
      case 'maintenance':
        // 系统维护：抛出错误，停止重试
        throw new Error('网站正在维护中，请稍后再试');
        
      default:
        console.log(`检测到异常情况: ${anomaly}`);
    }
  }
  
  return { ok: false, reason: `检测到异常: ${anomalies.detected.join(', ')}` };
}
```

```javascript
/**
 * 鲁棒的查询流程
 */
async function robustQueryFlow(session, company, query, options) {
```
